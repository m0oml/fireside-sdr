# ft8_lib (MIT, vendored): C code, third-party warnings are not ours to fix
set(FT8_DIR ${CMAKE_SOURCE_DIR}/third_party/ft8_lib)
file(GLOB FT8_SOURCES ${FT8_DIR}/ft8/*.c)
add_library(ft8lib STATIC ${FT8_SOURCES} ${FT8_DIR}/fft/kiss_fft.c ${FT8_DIR}/fft/kiss_fftr.c ${FT8_DIR}/common/monitor.c)
target_include_directories(ft8lib PUBLIC ${FT8_DIR})
target_compile_options(ft8lib PRIVATE -w -O2 -include ${CMAKE_SOURCE_DIR}/third_party/ft8_quiet.h)
target_link_libraries(ft8lib PUBLIC m)
# NR2 (WDSP's EMNR, GPL v2+, vendored): needs the system libfftw3 (runtime library; the header is bundled). Optional: without it the app builds and NR2 is simply absent.
find_library(FFTW3_LIBRARY NAMES fftw3 libfftw3.so.3)
if(FFTW3_LIBRARY)
  set(NR2_DIR ${CMAKE_SOURCE_DIR}/third_party/wdsp-nr2)
  add_library(wdsp_nr2 STATIC ${NR2_DIR}/src/emnr.c ${NR2_DIR}/src/calculus.c ${NR2_DIR}/src/zetaHat.c ${NR2_DIR}/src/fireside_emnr.c)
  target_include_directories(wdsp_nr2 PRIVATE ${NR2_DIR}/include ${NR2_DIR}/src)
  target_compile_definitions(wdsp_nr2 PRIVATE WDSP_STANDALONE)
  target_compile_options(wdsp_nr2 PRIVATE -w -O2)
  set_target_properties(wdsp_nr2 PROPERTIES POSITION_INDEPENDENT_CODE ON C_STANDARD 11)
  target_link_libraries(wdsp_nr2 PUBLIC ${FFTW3_LIBRARY} m)
  message(STATUS "NR2 (WDSP EMNR) enabled: ${FFTW3_LIBRARY}")
else()
  message(STATUS "libfftw3 not found: NR2 (spectral noise reduction) not built")
endif()

target_include_directories(fireside_core PUBLIC include)
target_compile_features(fireside_core PUBLIC cxx_std_17)
